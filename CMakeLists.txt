# Native GoldSource observer, launched explicitly for local/co-op games. The $LIB
# loader token handles both Arch and Debian/Steam runtime directory conventions.
option(G13PAD_BUILD_GOLDSRC_HEALTH "Build native GoldSource health handler" ON)
if(G13PAD_BUILD_GOLDSRC_HEALTH)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-m32")
  set(CMAKE_REQUIRED_LINK_OPTIONS "-m32")
  check_cxx_source_compiles("int main() { return 0; }" G13PAD_HAVE_M32)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  set(goldsrc_archs x86_64)
  if(G13PAD_HAVE_M32)
    list(APPEND goldsrc_archs i386)
  else()
    message(WARNING "No 32-bit toolchain: stock GoldSource requires the i386 observer")
  endif()
  foreach(arch IN LISTS goldsrc_archs)
    add_library(goldsrc-${arch} MODULE contrib/goldsrc-health/goldsrc.cpp)
    set_target_properties(goldsrc-${arch} PROPERTIES PREFIX "" SUFFIX ".so")
    target_compile_options(goldsrc-${arch} PRIVATE -O2 -fno-exceptions -fno-rtti
      -fno-stack-protector -fvisibility=hidden -foptimize-sibling-calls)
    target_compile_definitions(goldsrc-${arch} PRIVATE "G13PAD_VERSION=\"${PROJECT_VERSION}\"")
    target_link_options(goldsrc-${arch} PRIVATE -Wl,-z,defs -Wl,--as-needed)
    target_link_libraries(goldsrc-${arch} PRIVATE -Wl,--no-as-needed -l:libdl.so.2 -l:librt.so.1 -Wl,--as-needed)
    if(arch STREQUAL "i386")
      target_compile_options(goldsrc-${arch} PRIVATE -m32)
      target_link_options(goldsrc-${arch} PRIVATE -m32)
      set(layouts lib32 lib/i386-linux-gnu)
    else()
      set(layouts lib lib64 lib/x86_64-linux-gnu)
    endif()
    foreach(layout IN LISTS layouts)
      install(FILES $<TARGET_FILE:goldsrc-${arch}>
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/g13pad/goldsrc-health/${layout}
        RENAME goldsrc.so COMPONENT goldsrc-health)
    endforeach()
    if(BUILD_TESTING)
      add_library(fake-goldsrc-${arch} MODULE tools/fake-goldsrc.cpp)
      set_target_properties(fake-goldsrc-${arch} PROPERTIES PREFIX "" OUTPUT_NAME client
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fake-goldsrc-${arch}")
      add_executable(check-goldsrc-${arch} tools/check-goldsrc.cpp)
      target_compile_options(check-goldsrc-${arch} PRIVATE -UNDEBUG)
      target_link_libraries(check-goldsrc-${arch} PRIVATE ${CMAKE_DL_LIBS})
      add_library(fake-goldsrc-next-${arch} MODULE tools/fake-goldsrc-lookup.cpp)
      set_target_properties(fake-goldsrc-next-${arch} PROPERTIES PREFIX "")
      add_library(fake-goldsrc-interposer-${arch} MODULE tools/fake-goldsrc-interposer.cpp)
      target_compile_options(fake-goldsrc-interposer-${arch} PRIVATE -O2 -fno-exceptions -foptimize-sibling-calls)
      target_link_libraries(fake-goldsrc-interposer-${arch} PRIVATE ${CMAKE_DL_LIBS})
      # SHARED permits a dependent test library; MODULE cannot be linked to.
      add_library(fake-goldsrc-scope-second-${arch} SHARED tools/fake-goldsrc-lookup.cpp)
      target_compile_definitions(fake-goldsrc-scope-second-${arch} PRIVATE G13_SECOND)
      target_link_libraries(fake-goldsrc-next-${arch} PRIVATE -Wl,--no-as-needed
        fake-goldsrc-scope-second-${arch} -Wl,--as-needed ${CMAKE_DL_LIBS})
      if(arch STREQUAL "i386")
        foreach(t fake-goldsrc-i386 check-goldsrc-i386 fake-goldsrc-next-i386 fake-goldsrc-scope-second-i386 fake-goldsrc-interposer-i386)
          target_compile_options(${t} PRIVATE -m32)
          target_link_options(${t} PRIVATE -m32)
        endforeach()
      endif()
      foreach(route named table)
        if(route STREQUAL "table")
          set(extra table)
        else()
          set(extra)
        endif()
        add_test(NAME goldsrc-health-${route}-${arch}
          COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:goldsrc-${arch}>"
          $<TARGET_FILE:check-goldsrc-${arch}> $<TARGET_FILE:fake-goldsrc-${arch}> ${extra})
        add_test(NAME goldsrc-health-chain-${route}-${arch}
          COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:goldsrc-${arch}>:$<TARGET_FILE:fake-goldsrc-interposer-${arch}>"
          $<TARGET_FILE:check-goldsrc-${arch}> $<TARGET_FILE:fake-goldsrc-${arch}> ${extra})
      endforeach()
      add_test(NAME goldsrc-health-lookup-${arch}
        COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:goldsrc-${arch}>"
        $<TARGET_FILE:check-goldsrc-${arch}> --lookup $<TARGET_FILE:fake-goldsrc-next-${arch}>)
      add_test(NAME goldsrc-health-chain-lookup-${arch}
        COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:goldsrc-${arch}>:$<TARGET_FILE:fake-goldsrc-interposer-${arch}>"
        $<TARGET_FILE:check-goldsrc-${arch}> --lookup $<TARGET_FILE:fake-goldsrc-next-${arch}>)
      add_test(NAME goldsrc-health-symbols-${arch}
        COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/tools/check-goldsrc-symbols.sh" $<TARGET_FILE:goldsrc-${arch}>)
    endif()
  endforeach()
  configure_file(contrib/goldsrc-health/launch.sh.in g13map-goldsrc @ONLY)
  install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/g13map-goldsrc"
    DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT goldsrc-health)
endif()

# The Source engine's LCD module (contrib/source-health): dropped into a game's bin/ and
# loaded by the client itself with -g15, so it carries no libstdc++ and no glibc symbol
# newer than Steam's runtimes. Defined before the sanitizer block: instrumented code could
# not load inside the game.
option(G13PAD_BUILD_SOURCE_HEALTH "Build the Source engine health feeder modules" ON)
if(G13PAD_BUILD_SOURCE_HEALTH)
  set(g15_dir "${CMAKE_CURRENT_SOURCE_DIR}/contrib/source-health")
  set(g15_compile -O2 -fno-exceptions -fno-rtti -fno-stack-protector -fvisibility=hidden
      -fno-asynchronous-unwind-tables)
  set(g15_link -nostdlib++ -static-libgcc -Wl,-z,defs -Wl,--as-needed)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-m32")
  set(CMAKE_REQUIRED_LINK_OPTIONS "-m32")
  check_cxx_source_compiles("int main() { return 0; }" G13PAD_HAVE_M32)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  set(g15_archs x86_64)
  if(G13PAD_HAVE_M32)
    list(APPEND g15_archs i386)   # Half-Life 2 and the other 32-bit clients
  else()
    message(WARNING "No 32-bit toolchain: the i386 modules (Half-Life 2 and other 32-bit Source games) are not built")
  endif()
  # mp: the feeder alone, for VAC-secured multiplayer games. sp: plus the Battery
  # user-message hook for the shield bar (a vtable patch, singleplayer only).
  foreach(g15_arch IN LISTS g15_archs)
    foreach(g15_variant mp sp)
      set(g15_target g15-${g15_variant}-${g15_arch})
      if(g15_variant STREQUAL "sp")
        add_library(${g15_target} MODULE "${g15_dir}/g15.cpp" "${g15_dir}/armour.cpp")
        target_compile_definitions(${g15_target} PRIVATE G13_SINGLEPLAYER)
        target_link_libraries(${g15_target} PRIVATE ${CMAKE_DL_LIBS})
      else()
        add_library(${g15_target} MODULE "${g15_dir}/g15.cpp")
      endif()
      set_target_properties(${g15_target} PROPERTIES PREFIX "" SUFFIX ".so")
      target_compile_options(${g15_target} PRIVATE ${g15_compile})
      target_compile_definitions(${g15_target} PRIVATE "G13PAD_VERSION=\"${PROJECT_VERSION}\"")
      target_link_options(${g15_target} PRIVATE ${g15_link})
      if(g15_arch STREQUAL "i386")
        target_compile_options(${g15_target} PRIVATE -m32)
        target_link_options(${g15_target} PRIVATE -m32)
      endif()
      install(TARGETS ${g15_target} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/g13pad/source-health)
    endforeach()
  endforeach()
  install(FILES contrib/source-health/g15.res DESTINATION ${CMAKE_INSTALL_DATADIR}/g13pad/source-health)
  if(BUILD_TESTING)
    foreach(g15_arch IN LISTS g15_archs)
      add_executable(check-g15-${g15_arch} tools/check-g15.cpp)
      target_link_libraries(check-g15-${g15_arch} PRIVATE ${CMAKE_DL_LIBS})
      # The stand-in client library the sp module patches; its name is what the hook looks for.
      add_library(fake-client-${g15_arch} MODULE tools/fake-client.cpp)
      set_target_properties(fake-client-${g15_arch} PROPERTIES PREFIX "" OUTPUT_NAME client SUFFIX ".so"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fake-client-${g15_arch}")
      if(g15_arch STREQUAL "i386")
        foreach(t check-g15-i386 fake-client-i386)
          target_compile_options(${t} PRIVATE -m32)
          target_link_options(${t} PRIVATE -m32)
        endforeach()
      endif()
      foreach(g15_variant mp sp)
        set(g15_target g15-${g15_variant}-${g15_arch})
        if(g15_variant STREQUAL "sp")
          add_test(NAME source-health-load-${g15_variant}-${g15_arch}
            COMMAND check-g15-${g15_arch} $<TARGET_FILE:${g15_target}> $<TARGET_FILE:fake-client-${g15_arch}>)
        else()
          add_test(NAME source-health-load-${g15_variant}-${g15_arch}
            COMMAND check-g15-${g15_arch} $<TARGET_FILE:${g15_target}>)
        endif()
        add_test(NAME source-health-symbols-${g15_variant}-${g15_arch}
          COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/tools/check-g15-symbols.sh" $<TARGET_FILE:${g15_target}>)
        set_tests_properties(source-health-symbols-${g15_variant}-${g15_arch} PROPERTIES SKIP_RETURN_CODE 77)
      endforeach()
    endforeach()
  endif()
endif()

if(G13PAD_BUILD_EDITOR OR G13PAD_BUILD_CLI OR G13PAD_BUILD_ADAPTER)
  find_program(CARGO_EXECUTABLE cargo REQUIRED)
  set(cargo_args --locked)
  if(G13PAD_CARGO_OFFLINE)
    list(APPEND cargo_args --offline)
  endif()
  # Cargo owns freshness (including include_bytes! assets and manifest changes).
  # Separate package selections keep adapter/driver work out of the GUI dependency graph.
  foreach(component cli editor anim obs factorio adapter)
    set(rust_features)
    set(rust_test_args)
    if(component STREQUAL "editor")
      set(rust_package g13map)
      set(rust_binary g13map-editor)
      set(rust_test rust-tests)
      set(component_enabled ${G13PAD_BUILD_EDITOR})
    elseif(component STREQUAL "anim")
      # The built-in LCD animations; the editor's Animations window shares their code.
      set(rust_package g13map)
      set(rust_binary g13map-anim)
      set(rust_test anim-tests)
      set(component_enabled ${G13PAD_BUILD_EDITOR})
      list(APPEND rust_features --no-default-features --features art)
    elseif(component STREQUAL "obs")
      # The OBS overlay window; it shares the editor's GUI dependencies.
      set(rust_package g13map)
      set(rust_binary g13map-obs)
      set(rust_test obs-tests)
      set(component_enabled ${G13PAD_BUILD_EDITOR})
    elseif(component STREQUAL "factorio")
      set(rust_package g13map)
      set(rust_binary g13map-factorio)
      set(rust_test factorio-tests)
      list(APPEND rust_test_args --bin g13map-factorio)
      set(component_enabled ${G13PAD_BUILD_CLI})
      list(APPEND rust_features --no-default-features)
    elseif(component STREQUAL "cli")
      set(rust_package g13map)
      set(rust_binary g13map)
      set(rust_test cli-tests)
      set(component_enabled ${G13PAD_BUILD_CLI})
      list(APPEND rust_features --no-default-features)
    else()
      set(rust_package g13pad-analog)
      set(rust_binary g13pad-analog)
      set(rust_test adapter-tests)
      set(component_enabled ${G13PAD_BUILD_ADAPTER})
    endif()
    if(NOT component_enabled)
      continue()
    endif()
    add_custom_target(${component} ALL
      COMMAND ${CARGO_EXECUTABLE} build -p ${rust_package}
        --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
        --target-dir "${G13PAD_CARGO_TARGET_DIR}" --profile ${G13PAD_RUST_PROFILE}
        --bin ${rust_binary} ${rust_features} ${cargo_args} -j ${G13PAD_RUST_JOBS}
      BYPRODUCTS "${rust_output_dir}/${rust_binary}"
      USES_TERMINAL VERBATIM)
    install(PROGRAMS "${rust_output_dir}/${rust_binary}" DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(BUILD_TESTING)
      add_test(NAME ${component}-version COMMAND "${rust_output_dir}/${rust_binary}" --version)
      set_tests_properties(${component}-version PROPERTIES PASS_REGULAR_EXPRESSION "${PROJECT_VERSION}")
      # The editor's rust-tests cover the lean CLI and the animations too.
      if(NOT component MATCHES "^(cli|anim)$" OR NOT G13PAD_BUILD_EDITOR)
        add_test(NAME ${rust_test} COMMAND ${CARGO_EXECUTABLE} test -p ${rust_package}
          --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
          --target-dir "${G13PAD_CARGO_TARGET_DIR}" --profile ${G13PAD_RUST_TEST_PROFILE}
          ${rust_features} ${rust_test_args} ${cargo_args} -j ${G13PAD_RUST_JOBS})
        # Application fixtures export process-global G13MAP_* variables.
        set_tests_properties(${rust_test} PROPERTIES TIMEOUT 300 ENVIRONMENT "RUST_TEST_THREADS=1")
      endif()
    endif()
  endforeach()
  if(BUILD_TESTING)
    add_test(NAME core-tests COMMAND ${CARGO_EXECUTABLE} test -p g13pad-core
      --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
      --target-dir "${G13PAD_CARGO_TARGET_DIR}" --profile ${G13PAD_RUST_TEST_PROFILE}
      ${cargo_args} -j ${G13PAD_RUST_JOBS})
    set_tests_properties(core-tests PROPERTIES TIMEOUT 60)
  endif()
endif()

# The OBS source plugin: the pad as a source beside the input-overlay ones (obs/).
option(G13PAD_BUILD_OBS "Build the OBS source plugin (needs libobs and jansson)" ON)
if(G13PAD_BUILD_OBS)
  find_package(libobs QUIET)
  pkg_check_modules(JANSSON QUIET IMPORTED_TARGET jansson)
  if(libobs_FOUND AND JANSSON_FOUND)
    add_library(g13pad-obs MODULE obs/g13pad-obs.c)
    set_target_properties(g13pad-obs PROPERTIES PREFIX "" C_STANDARD 11 C_EXTENSIONS ON)
    target_link_libraries(g13pad-obs PRIVATE OBS::libobs PkgConfig::JANSSON m)
    target_compile_definitions(g13pad-obs PRIVATE
      G13PAD_OBS_SHEET_DIR="${CMAKE_INSTALL_FULL_DATADIR}/g13pad/obs"
      G13PAD_VERSION="${PROJECT_VERSION}")
    install(TARGETS g13pad-obs LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
    install(DIRECTORY obs/data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/obs/obs-plugins/g13pad-obs)
    if(BUILD_TESTING)
      add_test(NAME obs-plugin-symbols COMMAND sh -c "nm -D --defined-only '$<TARGET_FILE:g13pad-obs>' | grep -q ' T obs_module_load'")
    endif()
  else()
    message(STATUS "OBS source plugin not built: libobs or jansson not found")
  endif()
endif()
